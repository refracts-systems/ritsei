<script type="text/x-dc" data-dc-script data-props='{"$preview":{"width":1440,"height":1000}}'>
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { sel: 'appr', cfo: false, linked: true };
  }
  renderVals() {
    const { sel, cfo, linked } = this.state;
    const col = { cmd: '#2E5A87', cond: '#8A5A12', appr: '#B5402B', human: '#62676D' };
    const all = [
      { id: 'so', kind: 'Perintah', t: 'cmd', title: 'Sales Order', x: 30, y: 20, rule: 'Sales order dibuat dari penawaran yang sudah disetujui pelanggan.', contract: 'SalesOrder.create' },
      { id: 'kredit', kind: 'Kondisi', t: 'cond', title: 'Cek Kredit', x: 30, y: 150, rule: 'Jika total order ditambah piutang melebihi batas kredit, ikuti cabang gagal.', contract: 'Credit.check' },
      { id: 'review', kind: 'Tugas manusia', t: 'human', title: 'Review Manual', x: 400, y: 150, rule: 'Analis kredit meninjau dalam satu hari kerja, lalu alur kembali ke approval.', contract: 'Task.review' },
      { id: 'appr', kind: 'Approval', t: 'appr', title: 'Approval Manajer', x: 30, y: 280, rule: 'Jika nilai SO di atas Rp500 jt, tambahkan approval CFO.', contract: 'SalesOrder.approve' },
      { id: 'stok', kind: 'Perintah', t: 'cmd', title: 'Cadangkan Stok', x: 30, y: 410, rule: 'Cadangkan stok untuk baris yang tersedia di gudang asal.', contract: 'Stock.reserve' },
      { id: 'faktur', kind: 'Perintah', t: 'cmd', title: 'Terbitkan Faktur', x: 30, y: 540, rule: 'Terbitkan faktur dari barang yang sudah dikirim, dengan harga dan pajak dari order.', contract: 'Invoice.issue' }
    ];
    if (cfo) all.push({ id: 'cfo', kind: 'Approval', t: 'appr', title: 'Approval CFO', x: 640, y: 280, rule: 'Approval CFO untuk SO di atas Rp500 jt, sebelum stok dicadangkan.', contract: 'SalesOrder.approveCfo' });
    const nodes = all.map((n) => ({
      ...n, color: col[n.t], sel: sel === n.id, cls: sel === n.id ? 'sel' : '',
      pick: () => this.setState({ sel: n.id })
    }));
    const cur = all.find((n) => n.id === sel) || all[3];
    const ok = linked;
    return {
      nodes, cfo, linked,
      pick: { title: cur.title, rule: cur.rule, contract: cur.contract },
      vshort: ok ? 'Validasi lolos' : 'Validasi gagal',
      vlong: ok ? 'Lolos: semua cabang berakhir di status final.' : 'Gagal: cabang Review Manual berhenti tanpa status final. Alur belum bisa dipublikasikan.',
      vcolor: ok ? '#2E6B4A' : '#A33925',
      vbg: ok ? '#2E6B4A' : '#A33925',
      cfoLabel: cfo ? 'Hapus approval CFO' : 'Tambahkan approval CFO',
      cutLabel: linked ? 'Putuskan Review Manual dari Approval' : 'Sambungkan kembali Review Manual',
      toggleCfo: () => this.setState({ cfo: !cfo, sel: !cfo ? 'cfo' : 'appr' }),
      toggleCut: () => this.setState({ linked: !linked, sel: 'review' })
    };
  }
}
