class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { q: 'PO', view: 'Normal', ran: false };
  }
  renderVals() {
    const { q, view, ran } = this.state;
    const colors = { FIN: '#7A2E3D', PRC: '#2E5A87', INV: '#4A6E4D', SAL: '#AE522C', Aksi: '#1B2638', Org: '#62676D' };
    const all = [
      { g: 'Objek', tag: 'PRC', code: 'PO-JKT-0412', what: 'Approval PO Rp 184,5 jt, Sinar Jaya', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Objek', tag: 'PRC', code: 'PR-JKT-0221', what: 'Permintaan 400 sak semen belum punya PO', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Objek', tag: 'FIN', code: 'INV-SBY-1093', what: 'Jatuh tempo besok, Rp 72 jt', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Objek', tag: 'SAL', code: 'SO-JKT-0412', what: 'PT Abadi Sentosa, Rp 412 jt', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Objek', tag: 'INV', code: 'SKU-0221', what: 'Stok 38 dari safety stock 120', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Aksi', tag: 'Aksi', code: 'Buka Yard', what: 'Lihat tumpukan Order-to-Cash', hint: 'layar', href: 'KYard.dc.html' },
      { g: 'Aksi', tag: 'Aksi', code: 'Rancang alur', what: 'Buka Process Studio, draf v3', hint: 'layar', href: 'KStudio.dc.html' },
      { g: 'Aksi', tag: 'Aksi', code: 'Tutup hari', what: 'Lihat ringkasan dan apa yang menunggu besok', hint: 'layar', href: 'KTutup.dc.html' },
      { g: 'Pihak', tag: 'Org', code: 'Sinar Jaya', what: 'Vendor, 6 PO tahun ini', hint: 'buka manifest', href: 'KManifest.dc.html' },
      { g: 'Pihak', tag: 'Org', code: 'PT Abadi Sentosa', what: 'Pelanggan, 31 order tahun ini', hint: 'buka manifest', href: 'KManifest.dc.html' }
    ];
    const needle = q.trim().toLowerCase();
    const hit = all.filter((x) => needle === '' || (x.code + ' ' + x.what + ' ' + x.tag + ' ' + x.g).toLowerCase().includes(needle));
    let first = true;
    const groups = ['Objek', 'Aksi', 'Pihak'].map((n) => ({
      name: n,
      items: hit.filter((x) => x.g === n).map((x) => {
        const cls = first ? 'first' : '';
        first = false;
        return { ...x, cls, color: colors[x.tag] };
      })
    })).filter((g) => g.items.length > 0);
    const isCmd = /^buat po/.test(needle);
    const normal = (view === 'Normal') && hit.length > 0;
    return {
      q,
      onQ: (e) => this.setState({ q: e.target.value, view: 'Normal', ran: false }),
      groups,
      cmd: {
        show: isCmd,
        title: 'Buat PO · vendor Sinar Jaya',
        detail: 'Dari permintaan PR-JKT-0221, 400 sak semen. Masuk antrean approval sebelum dikirim.',
        ready: !ran, ran
      },
      run: () => this.setState({ ran: true }),
      v: { loading: view === 'Memuat', error: view === 'Gagal', empty: (view === 'Kosong') || (view === 'Normal' && hit.length === 0), normal: view === 'Normal' && hit.length > 0 },
      retry: () => this.setState({ view: 'Normal' }),
      views: ['Normal', 'Memuat', 'Kosong', 'Gagal'].map((s) => ({ label: s, on: view === s, cls: view === s ? 'on' : '', pick: () => this.setState({ view: s }) }))
    };
  }
}
