class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { sel: 1, invoiced: false, size: 3 };
  }
  renderVals() {
    const { sel, invoiced, size } = this.state;
    const nodes = [
      { code: 'SO-0412', text: 'Disetujui. Dibuat dari penawaran QT-0377 pada 21 Nov, diskon 12% disetujui Budi H.', done: true },
      { code: 'DO-0981', text: 'Dikirim dari Gudang Cikarang: 2 dari 3 barang sudah berangkat, sisanya menunggu stok.', now: !invoiced },
      { code: 'FKT-1120', text: invoiced ? 'Faktur dibuat dari DO-0981: hanya barang yang sudah dikirim, harga dan pajak dari order.' : 'Masih draf. Akan dibuat dari barang yang sudah dikirim.', done: invoiced, now: invoiced },
      { code: 'BAYAR', text: 'Belum ada pembayaran. Muncul di sini begitu mutasi bank cocok dengan faktur.' }
    ];
    const states = ['Disetujui', invoiced ? '2 dari 3 dikirim' : '2 dari 3 dikirim', invoiced ? 'Terbit' : 'Draf', 'Belum'];
    const chain = nodes.map((n, i) => ({
      code: n.code, state: states[i],
      cls: (n.done ? 'done ' : '') + (n.now ? 'now ' : '') + (sel === i ? 'sel' : ''),
      sub: n.done ? '#B4BDC9' : n.now ? '#A33925' : '#4A5260',
      sel: sel === i,
      pick: () => this.setState({ sel: i })
    }));
    const timeline = [
      ...(invoiced ? [{ time: '24 Nov 10:30', what: 'Faktur FKT-1120 dibuat dari DO-0981 oleh Rina A.' }] : []),
      { time: '24 Nov 09:12', what: 'Surat jalan DO-0981 dikirim dari Gudang Cikarang' },
      { time: '22 Nov 15:40', what: 'Diskon 12% disetujui oleh Budi H. (Manajer Sales)' },
      { time: '21 Nov 10:05', what: 'SO dibuat dari penawaran QT-0377' }
    ];
    const labels = ['Titik', 'Strip', 'Kartu', 'Manifest'];
    const notes = ['Titik: ukuran paling kecil, untuk yard dengan ribuan objek.', 'Strip: satu baris, untuk dermaga.', 'Kartu: pratinjau di panel samping.', 'Manifest: halaman objek lengkap seperti di kiri.'];
    return {
      chain, timeline,
      pick: { code: nodes[sel].code, text: nodes[sel].text },
      status: invoiced ? 'Terkirim sebagian, difaktur' : 'Terkirim sebagian',
      canInvoice: !invoiced,
      makeInvoice: () => this.setState({ invoiced: true, sel: 2 }),
      sizes: labels.map((l, i) => ({ label: l, on: size === i, cls: size === i ? 'on' : '', pick: () => this.setState({ size: i }) })),
      lod: { titik: size === 0, strip: size === 1, kartu: size === 2, manifest: size === 3, note: notes[size] }
    };
  }
}
